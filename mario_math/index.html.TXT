    // Generate Random Question
    function nextQuestion() {
      const num1 = Math.floor(Math.random() * 20) + 1;
      const num2 = Math.floor(Math.random() * 20) + 1;
      currentCorrectAnswer = num1 + num2;

      document.getElementById('question').innerText = `${num1} + ${num2} = ?`;

      let options = [currentCorrectAnswer];
      while (options.length < 4) {
        let wrong = currentCorrectAnswer + (Math.floor(Math.random() * 10) - 5);
        if (wrong > 0 && !options.includes(wrong)) {
          options.push(wrong);
        }
      }
      options.sort(() => Math.random() - 0.5);

      const optionsGrid = document.getElementById('options');
      optionsGrid.innerHTML = '';
      options.forEach(opt => {
        const btn = document.createElement('button');
        btn.className = 'btn-option';
        btn.innerText = opt;
        btn.onclick = () => checkAnswer(opt);
        optionsGrid.appendChild(btn);
      });
    }

        endGame();
      } else {
        nextQuestion();
      }
    }

// 8-ბიტიანი ხმის გენერატორი (Web Audio API)
const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playRetroSound(freq, type, duration) {
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type; // 'square', 'sawtooth', 'triangle'
  osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
  gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

// 🔊 სწორი პასუხის ხმა (Mario Coin Sound)
function playCoinSound() {
  playRetroSound(987.77, 'square', 0.08); // B5
  setTimeout(() => playRetroSound(1318.51, 'square', 0.25), 80); // E6
}

function playMusic(trackName) {
  // ყველა მუსიკის გაჩერება
  Object.values(sounds).forEach(s => { s.pause(); s.currentTime = 0; });
  
  // არჩეული ტრეკის ჩართვა
  if (sounds[trackName]) {
    sounds[trackName].play().catch(e => console.log("User interaction required to play audio"));
  }
}
